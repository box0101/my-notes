1. **防抖&节流**

   - **防抖**是在事件触发停止后延迟执行一次，用户每触发一次就会重置定时器重新计时，只关注最后一次触发。

     **使用场景：**输入框搜索、表单校验
     
     ```javascript
     function debounce(fn, delay) {
         let timer = null
         return function(...args) {
             if(timer) clearTimeout(timer)
             timer = setTimeout(() => {
                 fn.apply(this, args)
             }, delay)
         }
     }
     
     // 升级版：允许通过 immediate 设置立即执行，并且提供取消的方法。
     function debounce(fn, delay, immediate = false) {
       let timer = null
     
       function debounced(...args) {
         const context = this
     
         const callNow = immediate && !timer
     
         if (timer) clearTimeout(timer)
     
         timer = setTimeout(() => {
           timer = null
           if (!immediate) {
             fn.apply(context, args)
           }
         }, delay)
     
         if (callNow) {
           fn.apply(context, args)
         }
       }
     
       debounced.cancel = function () {
         if (timer) {
           clearTimeout(timer)
           timer = null
         }
       }
     
       return debounced
     }
     ```
     
   - 节流是每隔一段时间就执行一次回调函数，目的是均匀控制执行频率，避免频繁触发。

     **使用场景：**滚动监听、鼠标拖拽

     ```javascript
     function throttle(fn, delay) {
         let lastTime = 0;
         return function(...args) {
             const now = Date.now();
             if(now - lastTime <= delay) {
                 lastTime = Date.now();
                 fn.apply(this, args);
             }
         }
     }
     ```

5. **发布-订阅模式**

```javascript
class eventEmitters = {
    constrctor() {
        this.events = {}
    }
    
    // 订阅
    on(eventName, callback) {
        if(!this.events[eventName]) {
            this.events[eventName] = []
        }
        events[eventName].push(callback)
        return this
    }

	// 发布事件
	emit(eventName, callback) {
        if(!this.events[eventName]) {
            return this
        }
        const callbacks = [...this.events[eventName]]
        callbacks.forEach(callback => {
            callback.apply(this, args)
        })
        return this
    }

	// 取消订阅
	off(eventName, callback) {
        if(!this.events[eventName]) {
            return this
        }
        this.events[eventName].filter(cb => {
            cb !== callback && cb.original !== callback
        })
        return this
    }

	// 单次订阅
	once(eventName, callback) {
        const onceWrapper = (...args) => {
            callback.apply(this, args)
            this.off(eventName, onceWrapper)
        }
        
        onceWrapper.original = callback
        
        this.on(eventName, callback)
        return this
    }
}
```

7. **实现一个简单的任务重试机制** 

   接收一个返回 Promise 的异步函数 `fn`。 

   接收一个重试次数 retries。

   如果 `fn` 执行成功，直接返回结果。 

   如果执行失败，等待一段时间后进行重试，直到达到 retries 次数；如果最后一次依然失败，则抛出异常。

   ```javascript
   function sleep(delay) {
       return new Promise(resolve => setTimeout(resolve, delay));
   }
   
   async function pRetry(fn, retries, delay = 300) {
       let count = 0;
       
       while(count <= retries) {
           try {
               const res = await fn();
               return res;
           } catch(err) {
               if(count === retries) {
                   throw err;
               }
               await sleep(delay);
               count++;
           }
       }
   }
   ```

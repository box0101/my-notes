##### 1. **下一个更大的数（单调栈）**

**Next Greater Element I（单调栈+哈希表）**

（1）暴力解法

先在nums2中找到nums1的值，接着利用 k = j + 1 遍历数组右边，直到找到比目标值大的数，就将其赋值给res[i]

```js
var nextGreaterElements = function(nums1, nums2){
    const m = nums1.length, n = nums2.length
    const res = new Array(m).fill(0)
    
    for(let i = 0; i < m; i++){
        let j = 0
        while(j < n && nums1[i] !== nums2[j]){
            j++
        }
        let k = j + 1
        while(k < n && nums2[k] <= nums2[j]){
            k++
        }
        res[i] = k < n ? nums2[k] : -1
    }
    return res
}
```

（2）单调栈+哈希表

将nums2从左到右/从右到左遍历通过单调栈得到对应的第一个更大值，并以键值对形式写入哈希表，通过查表得到nums1对应第一个更大值。

```js
var nextGreaterElements = function(nums1, nums2){
    const map = new Map()
    const stack = []
    
    // 从左到右遍历 nums2
    for(let i = 0; i < nums2.length; i++){
        while(stack.length && stack[stack.length - 1] < nums2[i]){
            map.set(stack[stack.length - 1], nums2[i])
            stack.pop()
        }
        stack.push(nums2[i])
    }
    
    while(stack.length){
        map.set(stack.pop(), -1)
    }
    
    // 从右到左遍历
    for(let i = nums2.length - 1; i >= 0; i--){
        while(stack.length && stack[stack.length - 1] <= nums2[i]){
            stack.pop()
        }
        map.set(nums2[i], stack.length ? stack[stack.length - 1] : -1)
        stack.push(nums2[i])
    }
    
    const res = nums1.map(num => map.get(num))
    return res
}
```



**Next Greater Element II（单调栈+循环数组）**

和 **I** 不同的是，此处数组可能有重复的值，所以栈内的值不能用数组的值表示，取而代之的是数组下标。同时，因为是循环数组所以需要对数组进行两次遍历。

```js
var nextGreaterElements = function(nums){
    const stack = []
    const n = nums.length
    const res = new Array(n).fill(-1)
    
    for(let i = 0; i < n * 2 - 1; i++){
        while(stack.length && nums[stack[stack.length - 1]] < nums[i % n]){
            res[stack[stack.length - 1]] = nums[i % n]
            stack.pop()
        }
        stack.push(i % n)
    }
    return res
}
```



**总结**

这类题目考察的是**单调栈**。当从左到右遍历数组时，栈内是还未找到 NEG 的值，通过遍历数组比较找到

它的 NEG，而从右往左遍历时栈内是所有比当前 nums[i] 大的数，栈顶元素则是 NEG，如果栈为空说明无所求。

单调栈问题思考三个关键点（从左到右遍历数组）：

1. 存什么？栈内一般存索引/数值，可以根据数组是否会有重复值来考虑。
2. 什么顺序？
   - 找下一个更大：栈内保持递减
   - 找下一个更小：栈内保持递增
3. 何时出栈？破坏了栈内条件就出栈，栈顶为所求。

##### 2. **除自身外的数组乘积（前后缀分解）**

（1）左右列表乘积（空间复杂度 O(N)）

两个数组分别记录数组每个值的左边所有数乘积和右边所有数乘积，两个数组相乘得到所求。

```javascript
var productExceptSelf = function(nums){
    const n = nums.length
    
    const pre = new Array(n)
    pre[0] = 1
    for(let i = 1; i < n; i++){
        pre[i] = pre[i - 1] * nums[i - 1]
    }
    
    const suf = new Array(n)
    suf[n - 1] = 1
    for(let i = n - 2; i >= 0; i--){
        suf[i] = suf[i + 1] * nums[i + 1]
    }
    
    const res = new Array(n)
    for(let i = 0; i < n; i++){
        res[i] = pre[i] * suf[i]
    }
    
    return res
}
```



（2）方法一优化空间复杂度为 O(1)

只用一个 pre 数组记录左边的乘积，直接用一个遍历更新与右边元素的乘积得到结果。

```javascript
var productExceptSelf = function(nums){
    const n = nums.length
    const res = new Array(n).fill(1)
    
    for(let i = 1; i < n; i++){
        res[i] = res[i - 1] * nums[i - 1]
    }
    
    let right = 1
    for(let i = n - 1; i >= 0; i--){
        res[i] *= right
        right *= nums[i]
    }
    
    return res
}
```

##### 4. **无重复字符的最长子串（滑动窗口）**

用 Set 数据结构表示窗口，存放窗口内的字符。遍历字符串每个字符，通过 right 表示右边界，判断窗口内是否有重复字符，while 直到出现重复字符就把长度存到 res，每次遍历都取大值。

```javascript
var lengthOfLongestSubstring = function(s){
    const occ = new Set()
    const n = s.length
    
    let right = -1, res = 0
    for(let i = 0; i < n; i++){
        if(i > 0){
            occ.delete(s.charAt(i - 1))
        }
        while(right + 1 < n && !occ.has(s.charAt(right + 1))){
            occ.add(s.charAt(right + 1))
            right++
        }
        res = Math.max(res, right - i + 1)
    }
    return res
}
```

##### 8. **三数之和（双指针）**

```javascript
var threeSum = function(nums) {
    const res = []
    const n = nums.length
    
    nums.sort((a, b) => a - b)
    if(nums == null || n < 3) return res
    for(let i = 0; i < n; i++) {
        if(nums[i] > 0) break
        if(i > 0 && nums[i] === nums[i - 1]) continue
        
        let left = i + 1, right = n - 1
        while(left < right) {
            let sum = nums[i] + nums[left] + nums[right]
            if(sum === 0) {
                res.push([nums[i], nums[left], nums[right]])
                while(left < right && nums[left] === nums[left + 1]) {
                    left++
                }
                while(left < right && nums[right] === nums[right -1]) {
                    right--
                }
                left++
                right--
            }else if(sum < 0) {
                left++
            }else {
                right--
            }
        }
    }
    return res
};
```

##### 13. **最大子数组和（动态规划+贪心算法）**

解题核心是动态规划，遍历数组，判断前序累计的和是否会对总和带来增益，如果前缀和加上当前值比单独当前值更大就说明是有收益的，保留，否则就丢弃前缀和只留下当前值。每走一步都要和全局的最大和进行比较留下更大的值。

```javascript
var maxSubArray = function(nums) {
    let res = nums[0]
    let pre = 0
    for(let x of nums) {
        pre = Math.max(pre + x, x)
        res = Math.max(res, pre)
    }
    return res
}
```

```javascript
var maxProfit = function(prices) {
    let n = prices.length
    const dp = new Array(n).fill([0, 0])
    
    dp[0] = [-prices[0], 0]
    for(let i = 1; i < n; i++) {
        dp[i] = [
            Math.max(dp[i - 1][0], -prices[i]),
            Math.max(dp[i - 1][1], prices[i] + dp[i - 1][0])
        ]
    }
    return dp[n - 1][1]
}
```

##### 16. 数组第k个最大元素（基于快排、堆排序的选择方法）

（1）快速排序（此处叫做快速选择更恰当）

一般的快速排序：随机选择一个元素作为基准，利用双指针将数组分为左右两堆（小于基准的和大于基准的），递归左右两堆完成排序。

此题可以返回升序排序后的第 n - k 个元素，但这样性能比较差，所以可以每次分堆完成后判断基准的下标是否等于目标数的下标，如果基准下标小于目标下标则继续在右边的数组里找，反之在左边数组里找。

```javascript
const partition = (nums, left, right) => {
    const index = left + Math.floor(Math.random() * (right - left + 1));
    const pivot = nums[index];
    [nums[index], nums[left]] = [nums[left], nums[index]];
    let i = left + 1, j = right;
    while(i <= j) {
        while(i <= j && nums[i] < pivot) i++;
        while(i <= j && nums[j] > pivot) j--;
        
        if(i > j) break;
        
        [nums[i], nums[j]] = [nums[j], nums[i]];
        i++;
        j--;
    }
    [nums[j], nums[left]] = [nums[left], nums[j]];
    return j
}

var findKthLargest = function(nums, k) {
    const n = nums.length;
    let i = 0, j = n - 1;
    
    while(true) {
        let index = partition(nums, i, j);
        if(index === n - k) {
            return nums[index];
        } else if (index > n - k) {
            j = index - 1;
        } else {
            i = index + 1;
        }
    }
}
```

（2）基于堆排序的选择方法

##### 18. 合并两个有序链表（递归/迭代）

（1）递归

可以如下递归地定义两个链表里的 merge 操作（忽略边界情况，比如空链表等）：

$$
merge(list_1, list_2) =
\begin{cases}
[list_1[0]] + merge(list_1[1:],\, list_2), & list_1[0] < list_2[0] \\\\
[list_2[0]] + merge(list_1,\, list_2[1:]), & otherwise
\end{cases}
$$

```javascript
var mergeTwoLists = function(list1, list2) {
    if(!list1) {
        return list2
    } else if(!list2) {
        return list1
    } else if(list1.val > list2.val) {
        list1.next = mergeTwoLists(list1.next, list2)
        return list1
    } else {
        list2.next = mergeTwoLists(list2.next, list1)
        return list2
    }
}
```

（2）迭代

##### 19. 螺旋矩阵（数组）

没有什么技巧就是模拟旋转的过程，考察怎么计算下一个坐标。

```javascript
var spiralOrder(matrix) {
    if(!matrix.length || !matrix[0].length) return [];
    
    const rows = matrix.length, cols = matrix[0].length;
    const total = rows * cols;
    const visited = new Array(rows).fill(0).map(() => new Array(cols).fill(false));
    const res = [];
    
    let row = 0, col = 0, directionIndex = 0;
    const directions = [[0, 1], [1, 0], [0, -1], [-1, 0]];
    for(let i = 0; i < total; i++) {
        res[i] = matrix[row][col];
        visited[row][col] = true;
        let nextRow = row + directions[directionIndex][0];
        let nextCol = col + directions[directionIndex][1];
        if(nextRow < 0 || nextRow >= rows || nextCol < 0 || nextCol >= cols || visited[nextRow][nextCol]) {
            directionIndex = (directionIndex + 1) % 4;
        }
        row += directions[directionIndex][0];
        col += directions[directionIndex][1];
    }
    return res
}
```

##### 21. 二叉树的层序遍历（BFS）

维护一个队列记录每一层的节点，每遍历队内一个节点就把节点出队并把左右子树入队。

```javascript
var levelOrder = function(root) {
    if(!root) return [];
    const res = [];
    const q = new Queue();
    q.enqueue(root);
    
    while(!q.isEmpty()) {
        const vals = [];
        let n = q.size();
        while(n--) {
            let node = q.dequeue();
            vals.push(node.val);
            if(node.left) q.enqueue(node.left);
            if(node.right) q.enqueue(node.right);
        }
        res.push(vals);
    }
    return res;
}
```

拓展：**二叉树的右视图**就是层序遍历返回每层最右边的值。



##### 22. 二叉树的深度优先搜索 DFS（前序/中序/后序）

###### **（1）二叉树的中序遍历（递归/迭代）**

①递归

```javascript
var inorderTraversal = function(root) {
    const res = [];
    
    const inorder = (root) => {
        if(!root) {
            return;
        }
        inorder(root.left);
        res.push(root.val);
        inorder(root.right);
    }
    
    inorder(root);
    return res;
}
```

②迭代

```javascript
var inorderTraversal = function(root) {
    const res = [];
    const stack = [];
    
    while(root || stack.length) {
        while(root) {
            stack.push(root);
            root = root.left;
        }
        root = stack.pop();
        res.push(root.val);
        root = root.right;
    }
    
    return res;
}
```

###### （2）二叉树前序遍历

```javascript
var preorderTraversal = function(root) {
    const res = [];
    
    const preOrder = (root) => {
        if(!root) return res;
        
        res.push(root.val);
        preOrder(root.left);
        preOrder(root.right);
    }
    
    preOrder(root);
    return res;
}
```



##### 23. 两数相加

模拟数学上两数相加流程。

```javascript
var addTwoNumbers = function(l1, l2) {
    let head = null, tail = null;
    let carry = 0;
    
    while(l1 || l2) {
        let x = l1 ? l1.val : 0;
        let y = l2 ? l2.val : 0;
        let sum = x + y + carry;
        if(!head) {
            head = tail = new ListNode(sum % 10);
        } else {
            tail.next = new ListNode(sum % 10);
            tail = tail.next;
        }
        if(l1) {
            l1 = l1.next;
        }
        if(l2) {
            l2 = l2.next;
        }
        carry = Math.floor(sum / 10);
    }
    if(carry > 0) {
        tail.next = new ListNode(carry);
    }
    return head;
}
```
